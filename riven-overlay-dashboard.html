// Example grading logic
function gradeRiven(riven, rivenData) {
  const weapon = rivenData.weapons[riven.weaponName];
  let totalScore = 0;

  for (const stat of riven.stats) {
    const weight = weapon.stats[stat.name]?.weight || 1;
    const disposition = weapon.disposition;
    totalScore += stat.value * weight * disposition;
  }

  return scoreToGrade(totalScore); // S/A/B/C
}
